namespace = egyptian_crisis_events

egyptian_crisis_events.1 = {
	type = country_event
	placement = scope:egyptian_crisis_state

	title = egyptian_crisis_events.1.t
	desc = egyptian_crisis_events.1.d
	flavor = egyptian_crisis_events.1.f

	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = c:EGY
		any_diplomatic_play = {
			initiator = root
			target = c:EGY
			OR = {
				has_play_goal = conquer_state
				has_play_goal = return_state
				has_play_goal = reduce_autonomy
				has_play_goal = make_tributary
				has_play_goal = make_protectorate
			}
		}
		NOT = {
			has_variable = egyptian_crisis
		}
	}

	immediate = {
		set_variable = {
			name = egyptian_crisis
		}
		if = {
			limit = { exists = s:STATE_SYRIA.region_state:EGY }
			s:STATE_SYRIA.region_state:EGY = { save_scope_as = egyptian_crisis_state }
		}
		else = {
			c:EGY.capital = { save_scope_as = egyptian_crisis_state }
		}
	}

	option = {
		name = egyptian_crisis_events.1.a
		add_modifier = {
			name = working_with_the_great_powers
			days = long_modifier_time
		}
		set_variable = egyptian_crisis_conciliatory
		every_country = {
			limit = {
				country_rank = rank_value:great_power
				country_is_in_europe = yes
				NOR = {
					c:EGY = this
					c:TUR = this
					has_war_with = c:TUR
					has_war_with = c:EGY
					has_diplomatic_pact = {
						who = root
						type = rivalry
					}
				}
			}
			change_relations = {
				country = root
				value = 30
			}
		}
		every_country = {
			limit = {
				country_rank = rank_value:great_power
				country_is_in_europe = yes
				NOR = {
					has_interest_marker_in_region = region_arabic
					c:EGY = this
					c:TUR = this
					has_war_with = c:TUR
					has_war_with = c:EGY
				}
			}
			hidden_effect = { trigger_event = { id = egyptian_crisis_events.4 } }
			custom_tooltip = egyptian_crisis_events.1.a.tt
		}
		c:EGY = {
			trigger_event = { id = egyptian_crisis_events.2 days = 5 popup = yes }
		}
		
		ai_chance = {
			base = 2
		}
	}

    option = {
		name = egyptian_crisis_events.1.b
        default_option = yes
		change_infamy = 5
		set_variable = egyptian_crisis_hardline
        every_country = {
        	limit = {
        		country_rank = rank_value:great_power
        		country_is_in_europe = yes
        		NOR = {
        			has_interest_marker_in_region = region_arabic
        			c:EGY = this
        			c:TUR = this
        			has_war_with = c:TUR
        			has_war_with = c:EGY
        		}
        	}
        	hidden_effect = { trigger_event = { id = egyptian_crisis_events.4 } }
        	custom_tooltip = egyptian_crisis_events.1.a.tt
        }
        c:EGY = {
	        	trigger_event = { id = egyptian_crisis_events.2 days = 5 popup = yes }
        }
		
		ai_chance = {
			base = 1
		}
	}
}

egyptian_crisis_events.4 = {
	type = country_event
	placement = scope:egyptian_crisis_state

	flavor = egyptian_crisis_events.4.f

	event_image = {
		video = "middleeast_courtroom_upheaval"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	immediate = {
		if = {
			limit = { exists = s:STATE_SYRIA.region_state:EGY }
			s:STATE_SYRIA.region_state:EGY = { save_scope_as = egyptian_crisis_state }
		}
		else = {
			c:EGY.capital = { save_scope_as = egyptian_crisis_state }
		}
	}

	option = {
		name = egyptian_crisis_events.4.a
        default_option = yes
		custom_tooltip = egyptian_crisis_events.4.a.tt
		hidden_effect = {
			if = {
				limit = {
					has_variable = veiled_protectorate_support_egypt
				}
				random_list = {
					50 = {}
					40 = {
						remove_variable = veiled_protectorate_support_egypt
					}
					10 = {
						remove_variable = veiled_protectorate_support_egypt
						set_variable = veiled_protectorate_support_ottomans
					}
				}
			}
			else_if = {
				limit = {
					NOT = { has_variable = veiled_protectorate_support_ottomans }
				}
				random_list = {
					50 = {}
					50 = {
						set_variable = veiled_protectorate_support_ottomans
					}
				}
			}
		}
		
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					c:TUR = {
						has_modifier = working_with_the_great_powers
					}
				}
				add = 2
			}
			modifier = {
				trigger = {
					c:TUR = {
						infamy <= infamy_threshold:infamous
					}
				}
				add = 1
			}
		}
	}

	option = {
		name = egyptian_crisis_events.4.b
		custom_tooltip = egyptian_crisis_events.4.b.tt
		hidden_effect = {
			if = {
				limit = {
					has_variable = veiled_protectorate_support_ottomans
				}
				random_list = {
					70 = {}
					20 = {
						remove_variable = veiled_protectorate_support_ottomans
					}
					10 = {
						remove_variable = veiled_protectorate_support_ottomans
						set_variable = veiled_protectorate_support_egypt
					}
				}
			}
		}
		
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					c:TUR = {
						infamy >= infamy_threshold:notorious
					}
				}
				add = 1
			}
			
			modifier = {
				trigger = {
					c:TUR = {
						infamy >= infamy_threshold:pariah
					}
				}
				add = 2
			}
		}
	}
}
