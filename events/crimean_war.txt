crimean_war.6 = {
	type = country_event
	placement = root
	title = crimean_war.6.t
	desc = crimean_war.6.d
	flavor = crimean_war.6.f
	
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/eu/diplomatic_play"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	trigger = {
		# Triggered by crimean_war.5
	}
	
	immediate = {
		c:RUS = {
			create_diplomatic_play = {
			type = dp_crimean_war
			target_country = c:TUR
			add_war_goal = {
				holder = c:TUR
				type = humiliation
				target_country = c:RUS
			}
		}
		random_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_crimean_war
				target = { this = c:TUR }
			}
			save_scope_as = crimean_war_play
		}
		}
	}

	option = { # War
		name = crimean_war.6.a
		
		if = { 
		    limit = {
			    exists = c:WAL
				c:WAL ?= {
					is_subject_of = c:TUR
				}
			}
			scope:crimean_war_play = { add_war_goal = { holder = c:RUS type = transfer_subject target_country = c:WAL } }
		}
		if = { 
		    limit = {
			    exists = c:MOL
				c:MOL ?= {
					is_subject_of = c:TUR
				}
			}
			scope:crimean_war_play = { add_war_goal = { holder = c:RUS type = transfer_subject target_country = c:MOL } }
		}
		if = { 
		    limit = {
			    exists = c:BUL
				c:BUL ?= {
					is_subject_of = c:TUR
				}
			}
			scope:crimean_war_play = { add_war_goal = { holder = c:RUS type = transfer_subject target_country = c:BUL } }
		}
		if = { 
		    limit = {
			    exists = c:ROM
				c:ROM ?= {
					is_subject_of = c:TUR
					
				}
			}
			scope:crimean_war_play = { add_war_goal = { holder = c:RUS type = transfer_subject target_country = c:ROM } }
		}
		
		if = { 
		    limit = {
			    exists = c:SER
				c:SER ?= {
					is_subject_of = c:TUR
					
				}
			}
			scope:crimean_war_play = { add_war_goal = { holder = c:RUS type = transfer_subject target_country = c:SER } }
		}
		
		c:FRA ?= {
			trigger_event = {
				id = crimean_war.7
				popup = yes
				days = 7
			}
		}
		c:SAR ?= {
			trigger_event = {
				id = crimean_war.7
				popup = yes
				days = 7
			}
		}
		c:GBR ?= {

			}
		}
	}

				every_scope_state = {
					limit = { 
						OR = {
							state_region = s:STATE_CHERSON
							state_region = s:STATE_CRIMEA
						}
						has_building = building_naval_base
					}
					remove_building = building_naval_base
				}
			}
		}
		
		custom_tooltip = {
			text = crimean_peace_moldavia_annexation_tt
			if = {
				limit = {
					exists = c:MOL
				}
				s:STATE_BESSARABIA = {
					set_owner_of_provinces = {
						country = c:MOL
						provinces = { x712161 x6bf7b8 xb0e0a0 }
					}
				}
			}
		}
		
		custom_tooltip = {
			text = crimean_peace_ottoman_annexation_tt
			if = {
				limit = {
					c:RUS = { owns_province = x50F5CB }
				}
				s:STATE_DOBRUDJA = {
					set_owner_of_provinces = {
						country = c:TUR
						provinces = { x50F5CB }
					}
				}
			}
		}
		
		# TODO: press western demands to forfeit ottoman control of christian vassals

		c:RUS = {
			add_modifier = {
				name = treaty_of_paris_demilitarized_black_sea
				months = 122
			}
		}
		c:TUR = {
			add_modifier = {
				name = treaty_of_paris_guaranteed_integrity
				months = 122
			}
		}
		
		c:RUS = {
			force_peace = yes
		}
		
		ai_chance = {
			base = 100
		}
	}
}
