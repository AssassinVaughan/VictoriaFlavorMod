namespace = liaotung_lease

liaotung_lease.1 = {
	type = country_event
	placement = ROOT

	trigger = {
		c:RUS ?= root
		exists = c:CHI
		c:CHI = {
			owns_entire_state_region = STATE_SHENGJING
		}
	}

	title = liaotung_lease.1.t
	desc = liaotung_lease.1.d
	flavor = liaotung_lease.1.f

	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	option = {
		name = liaotung_lease.1.a
		c:CHI ?= {
			trigger_event = {
				id = liaotung_lease.2
				days = 5
				popup = yes
			} 
		}
		custom_tooltip = liaotung_lease.1.a_tt
	}
	option = {
		name = liaotung_lease.1.b
		custom_tooltip = liaotung_lease.1.b_tt
	}
}

liaotung_lease.2 = {
	type = country_event
	placement = c:CHI

	trigger = {
		c:CHI ?= root
		exists = c:RUS
		owns_entire_state_region = STATE_SHENGJING
	}

	title = liaotung_lease.2.t
	desc = liaotung_lease.2.d
	flavor = liaotung_lease.2.f

	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	option = {
		name = liaotung_lease.2.a
		trigger = {
			c:RUS = {
				can_create_treaty = {
					first_country = c:RUS
					second_country = c:CHI

					is_draft = no
					binding_period = { years = 20 }

					articles_to_create = {
						{
							article = treaty_port
							source_country = c:CHI
							target_country = c:RUS
							inputs = {
								{ state = s:STATE_SHENGJING.region_state:CHI }
							}
						}
					}
				}
			}
		}
		custom_tooltip = liaotung_lease.2.a_tt
		create_treaty = {
			name = "Liaotung Lease"
			first_country = c:RUS
			second_country = c:CHI

			is_draft = no
			binding_period = {years = 20}

			articles_to_create = {
			{
				article = treaty_port
				source_country = c:CHI
				target_country = c:RUS
				inputs = {
					{ state = s:STATE_SHENGJING.region_state:CHI }
				}
			}
			}
		}
	}
	
	option = {
		name = liaotung_lease.2.b
		custom_tooltip = liaotung_lease.2.b_tt
		c:RUS ?= {
			trigger_event = {
				id = liaotung_lease.3
				days = 1
				popup = yes
			} 
		}
	}
}

liaotung_lease.3 = {
	type = country_event
	placement = c:RUS

	trigger = {
		c:RUS ?= root
		exists = c:CHI
		c:CHI = {
			owns_entire_state_region = STATE_SHENGJING
		}
	}

	title = liaotung_lease.3.t
	desc = liaotung_lease.3.d
	flavor = liaotung_lease.3.f

	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	option = {
		name = liaotung_lease.3.a
		trigger = {
			is_at_war = no
			c:CHI = {
				is_at_war = no
				owns_entire_state_region = STATE_SHENGJING
			}
		}
		custom_tooltip = liaotung_lease.3.a_tt
		create_diplomatic_play = {
			type = dp_take_treaty_port
			target = c:CHI
			target_state = s:STATE_SHENGJING.region_state:CHI
		}
	}
	option = {
		name = liaotung_lease.3.b
		custom_tooltip = liaotung_lease.3.b_tt
	}
}
