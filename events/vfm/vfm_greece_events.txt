vfm_greece.7 = {
	type = country_event
	placement = root

	title = vfm_greece.7.t
	desc = vfm_greece.7.d
	flavor = vfm_greece.7.f

	duration = 3

	event_image = {
		video = "votp_barricade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	cooldown = { days = normal_modifier_time }


	trigger = {
		# triggered by on_action
	}

	immediate = {
		cu:greek = {
			save_scope_as = greek_culture_scope
		}
		random_country = {
			limit = {
				has_variable = greek_rebel_country_var
			}
			save_scope_as = greek_rebel_country_scope
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:greek
					}
				}
				save_scope_as = greek_rebel_state_scope
			}
		}
		random_country = {
			limit = {
				has_variable = greek_rebel_country_enemy_var
			}
			save_scope_as = greek_rebel_country_enemy_scope
		}
		random_diplomatic_play = {
			limit = {
				target = {
					has_variable = greek_rebel_country_var
				}
			}
			save_scope_as = gre_diplo_play
		}
	}

	option = {
		name = vfm_greece.7.a
		add_radicals = {
			culture = cu:greek
			value = medium_radicals
		}
		change_relations = { country = scope:greek_rebel_country_scope value = -10 }
		change_relations = { country = scope:greek_rebel_country_enemy_scope value = 10 }
	}

	option = {
		name = vfm_greece.7.b
		highlighted_option = yes
		trigger = {
			is_subject = no
		}
		add_loyalists = {
			culture = cu:greek
			value = small_radicals
		}
		add_modifier = {
			name = funding_greek_rebels_small_modifier
			days = short_modifier_time
		}
		scope:greek_rebel_country_scope = {
			add_modifier = {
				name = greek_rebel_funding_modifier
				days = short_modifier_time
			}
		}
		change_relations = { country = scope:greek_rebel_country_scope value = 20 }
		change_relations = { country = scope:greek_rebel_country_enemy_scope value = -20 }
	}

	option = {
		name = vfm_greece.7.c
		highlighted_option = yes
		default_option = yes
		# TODO AI weight
		trigger = {
			is_subject = no
		}
		add_loyalists = {
			culture = cu:greek
			value = medium_radicals
		}
		add_modifier = {
			name = funding_greek_rebels_large_modifier
			days = short_modifier_time
		}
		scope:greek_rebel_country_scope = {
			add_modifier = {
				name = greek_rebel_funding_modifier
				days = short_modifier_time
				multiplier = 2
			}
		}
		change_relations = { country = scope:greek_rebel_country_scope value = 40 }
		change_relations = { country = scope:greek_rebel_country_enemy_scope value = -40 }
		

		scope:gre_diplo_play = {
			add_target_backers = { root }
		}
	}
}

				limit = { 
					any_scope_building = {
						is_building_type = building_barrack
					}
				}
			    remove_building = building_barrack
			}
	    }
	}
}
