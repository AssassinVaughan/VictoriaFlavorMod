journal_flavor_chile.3 = { #Colonization of Punta Arenas
	type = country_event
	placement = scope:araucania_state

	title = journal_flavor_chile.3.t
	desc = journal_flavor_chile.3.d
	flavor = journal_flavor_chile.3.f

	event_image = {
		video = "gfx/event_pictures/africa_construction_colony.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/africa/construction_colony"

	icon = "gfx/interface/icons/event_icons/event_flags/event_chl_flag.dds"

	duration = 1

	immediate = {
	    s:STATE_ARAUCANIA = {
            random_scope_state = {
                limit = {
                    owner = {
                        c:CHL = THIS
                    }
                }
                save_scope_as = araucania_state
            }
        }	
		c:CHL = {
			save_scope_as = chl_scope
		}
		s:STATE_ARAUCANIA = {
			set_owner_of_provinces = {
				country = c:CHL
				provinces = { x029AC8 x5A7430 }
			}
		}
		set_variable = magellan_strait	
		set_variable = magellan_strait1	
	}

	option = {
		default_option = yes
		name = journal_flavor_chile.3.a
        scope:araucania_state = {
            add_modifier = {
			    name = puq_penal_colony 
			    years = 20
		    }
	    }
    }
}

journal_flavor_chile.5 = { #War of the Pacific
	type = country_event
	placement = ROOT

	title = journal_flavor_chile.5.t
	desc = journal_flavor_chile.5.d
	flavor = journal_flavor_chile.5.f

	duration = 1

	event_image = {
		video = "gfx/event_pictures/middleeast_battlefield_trenches.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/middleeast/battlefield_trenches"

	icon = "gfx/interface/icons/event_icons/event_flags/event_chl_flag.dds"

	immediate = {
		c:BOL = {
			save_scope_as = bol_scope
		}
		c:CHL = {
			save_scope_as = chl_scope
		}
		c:PEU = {
			save_scope_as = peu_scope
		}
	    s:STATE_TARAPACA = {
            random_scope_state = {
                save_scope_as = tarapaca_state
            }
        }					
	}

	option = {
		default_option = yes
		name = journal_flavor_chile.5.a

		if = {
		    limit = {
		    	AND = {
		        	OR = {
		            	c:PEU = { has_global_variable = secret_alliance_var }
		            	has_global_variable = secret_alliance_var
		            }
		        }
		    }
            create_diplomatic_play = {
                name = war_of_the_pacific
                target_state = s:STATE_ANTOFAGASTA.region_state:BOL

                add_target_backers = { c:PEU }

                war = no

                type = dp_return_state

                add_war_goal = {
                    holder = c:CHL

                    type = conquer_state

                    target_state = scope:tarapaca_state
                }
            }
        }

		if = {
		    limit = {
		        NOT = { 
		        	AND = {
		                OR = {
		                	c:PEU = { has_global_variable = secret_alliance_var }
		                	has_global_variable = secret_alliance_var
		                }
		            }
		        }
		    }
            create_diplomatic_play = {
            	name = war_of_the_pacific
            	target_state = s:STATE_ANTOFAGASTA.region_state:BOL

            	war = no

            	type = dp_return_state
            }
        }

		add_loyalists = {
			value = 0.15
		}
					
		change_infamy = 5	

		hidden_effect = {
		    trigger_event = {
			    id = journal_flavor_chile.22
			    months = 1
		    }
		}
	}	        				
			            
	option = {
		name = journal_flavor_chile.5.b
    	c:CHL = {
        	add_modifier = {
				name = pacific_war_rejected
				months = normal_modifier_time
			}
			add_radicals = {
				value = 0.025	
			}
			add_radicals = {
				value = 0.4
				pop_type = capitalists	
			}
		}					
	}
}

journal_flavor_chile.6 = { #german immigration
	type = country_event
	placement = scope:osorno_state

	title = journal_flavor_chile.6.t
	desc = journal_flavor_chile.6.d
	flavor = journal_flavor_chile.6.f

	event_image = {
		video = "gfx/event_pictures/asia_poor_people_moving.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/poor_people_moving"

	icon = "gfx/interface/icons/event_icons/event_flags/event_chl_flag.dds"

	duration = 1

	immediate = {	
	    s:STATE_ARAUCANIA = {
            random_scope_state = {
                limit = {
                    owner = {
                        c:CHL = THIS
                    }
                }
                save_scope_as = osorno_state
            }
        }
		random_country = {
			limit = {
				country_rank <= rank_value:major_power
				average_sol <= root.average_sol
				OR = {
				    country_has_primary_culture = cu:north_german 
					country_has_primary_culture = cu:south_german
				}
				any_scope_state = {
					is_incorporated = yes
					custom_tooltip = {
						text = is_in_europe_tt
						state_is_in_europe = yes
					}
					count >= 3
				}
			}
			save_scope_as = migration_country
			random_scope_state = {
				limit = {
					is_incorporated = yes
					state_is_in_europe = yes
					any_scope_culture = {
						has_homeland = PREV
						culture_can_have_mass_migration_to_country = ROOT
					}
				}
				save_scope_as = migration_state
				random_scope_culture = {
					limit = {
						has_homeland = PREV
						culture_can_have_mass_migration_to_country = ROOT
						is_primary_culture = yes
					}
					save_scope_as = migration_culture
				}
			}
		}        
	}		

	option = {
		default_option = yes
		name = journal_flavor_chile.6.a

		scope:osorno_state = {
			create_mass_migration = {
				origin = scope:migration_country
				culture = scope:migration_culture
			}
		}
    }       
}
