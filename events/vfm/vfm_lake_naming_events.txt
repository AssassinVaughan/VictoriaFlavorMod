	placement = ROOT

	title = vfm_lake_naming.1.t

	duration = 3

	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	immediate = {
		vfm_lake_naming_scopes_effect = { CHAIN = nile }
	}

	#explorer
	option = {
		name = {
			trigger = { scope:vfm_explorer ?= { NOT = { has_variable = vfm_lake_namesake } } }
			text = vfm_lake_naming_explorer
		}
		name = {
			trigger = {
				scope:vfm_explorer ?= {
					has_variable = vfm_lake_namesake
					var:vfm_lake_namesake < 3
				}
			}
			text = vfm_lake_naming_explorer_again
		}
		name = {
			trigger = {
				scope:vfm_explorer ?= {
					has_variable = vfm_lake_namesake
					var:vfm_lake_namesake >= 3
				}
			}
			text = vfm_lake_naming_explorer_fourth
		}
		trigger = { exists = scope:vfm_explorer }
		default_option = yes
		vfm_lake_name_explorer_effect = { LAKE = victoria }
		ai_chance = {
			base = 3
			modifier = {
				trigger = { NOT = { has_law = law_type:law_monarchy } }
