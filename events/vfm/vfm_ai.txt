# 1838
vfm_ai.2 = {
	type = country_event
	placement = root

	hidden = yes

	duration = 1

	trigger = {
	    c:PRU ?= this
		#is_ai = yes
		game_date > 1837.7.1
		game_date < 1838.12.1
	}

	immediate = {
	    every_country = {
			limit = {
				is_ai = yes
                country_rank >= rank_value:minor_power
				has_technology_researched = railways
			}
			add_technology_progress = { progress = 4150 technology = steel_railway_cars } 
			add_modifier = {
				name = first_railway_modifier
				months = long_modifier_time
		    }
		}
		every_country = {
			limit = {
				is_ai = yes
                country_rank >= rank_value:minor_power
				has_technology_researched = atmospheric_engine
			}
			add_modifier = {
				name = atmospheric_engine_mining
			    months = normal_modifier_time
		    }
		}
	    every_country = {
			limit = {
				literacy_rate < 0.12
				is_ai = yes
			}
			every_scope_state = { # dont get stuck with butchering tools
			    limit = {
				has_building = building_livestock_ranch
				can_activate_production_method = {
					building_type = building_livestock_ranch
					production_method = pm_intensive_grazing_ranch
				}
				}
                activate_production_method = {
				building_type = building_livestock_ranch
				production_method = pm_intensive_grazing_ranch
		        }				
			}
		}
		
		if = {
			limit = {
			    c:PRU ?= {
					is_ai = yes
				}
			}
			c:PRU = {
				s:STATE_POMERANIA = {
					random_scope_state = {
						start_building_construction = building_port
					}
				}
				s:STATE_WESTPHALIA = {
					random_scope_state = {
						start_building_construction = building_tooling_workshops
						start_building_construction = building_livestock_ranch
					}
				}
			}
			c:MEC ?= {
		        set_owes_obligation_to = {
			    	country = c:PRU
			    	setting = yes
			    }
		    }
		}
		if = {
			limit = {
			    c:DEN ?= {
					is_ai = yes
				}
			}
			if = {
			limit = {
				any_scope_state = { # danes are obsessed with luxury furniture
						can_activate_production_method = {
							building_type = building_furniture_manufacturies
							production_method = pm_luxury_furniture
					    }	
					}
			}
			activate_production_method = {
				building_type = building_furniture_manufacturies
				production_method = pm_luxury_furniture
		    }
		}
		}
		if = {
			limit = {
			    c:USA ?= {
					is_ai = yes
				}
			}
			c:USA = {
				random_scope_state = {
					limit = {
						state_region = s:STATE_FLORIDA
						is_incorporated = no
					}
					# Civil war will be a mess without florida
					set_state_type = incorporated
				}
			}
		}
	}
}

vfm_ai.3 = {
	type = country_event
	placement = root

	hidden = yes

	duration = 1

	trigger = {
	    c:SWE ?= this
		is_ai = yes
		game_date > 1839.1.1
		game_date < 1840.6.1
	}

	immediate = {
	    if = {
			limit = {
				has_state_in_state_region = STATE_SCANIA
			}
			s:STATE_SCANIA = {
				random_scope_state = {
					start_building_construction = building_coal_mine
				}
			}
		}
		if = {
			limit = {
				any_scope_state = {
						can_activate_production_method = {
							building_type = building_iron_mine
							production_method = pm_condensing_engine_pump_building_iron_mine
					    }	
					}
			}
			activate_production_method = {
				building_type = building_iron_mine
				production_method = pm_condensing_engine_pump_building_iron_mine
		    }
		}
		else = {
		    activate_production_method = {
				building_type = building_iron_mine
				production_method = pm_atmospheric_engine_pump_building_iron_mine
		    }
		}
		
		if = {
			limit = {
				has_state_in_state_region = STATE_NORRLAND
			}
			s:STATE_NORRLAND = {
				random_scope_state = {
					start_building_construction = building_iron_mine
				}
			}
		}
	}
}

vfm_ai.4 = {
	type = country_event
	placement = root

	hidden = yes

	duration = 1

	trigger = {
		is_ai = yes
	}

	immediate = {
	    if = {
			limit = {
			    OR = {
			    AND = {
			        exists = c:CHI
					this = c:CHI
				}
				AND = {
			        exists = c:JAP
					this = c:JAP
				}
				}
				gold_reserves < 1000000 # 1 mil
				scaled_debt <= 0.35
			}
			set_government_wage_level = very_low
			set_military_wage_level = low
		}
	    else_ifif = {
			limit = {
				scaled_debt > 0.35
			}
			set_government_wage_level = very_low
			set_military_wage_level = very_low
		}
		else_if = {
			limit = {
				scaled_debt > 0.05
			}
			set_government_wage_level = low
			set_military_wage_level = low
		}
		else = {
		    set_government_wage_level = medium
			set_military_wage_level = medium
		}
	}
}
