vfm_ai.2 = {
	type = country_event
	placement = root

	hidden = yes

	duration = 1

	trigger = {
	    c:PRU ?= this
		#is_ai = yes
		game_date > 1837.7.1
		game_date < 1838.12.1
	}

	immediate = {
	    every_country = {
			limit = {
				is_ai = yes
                country_rank >= rank_value:minor_power
				has_technology_researched = railways
			}
			add_technology_progress = { progress = 4150 technology = steel_railway_cars } 
			add_modifier = {
				name = first_railway_modifier
				days = long_modifier_time
		    }
		}
		every_country = {
			limit = {
				is_ai = yes
                country_rank >= rank_value:minor_power
				has_technology_researched = atmospheric_engine
			}
			add_modifier = {
				name = atmospheric_engine_mining
			    days = normal_modifier_time
		    }
		}
	    every_country = {
			limit = {
				literacy_rate < 0.12
				is_ai = yes
			}
			every_scope_state = { # dont get stuck with butchering tools
			    limit = {
				has_building = building_livestock_ranch
				can_activate_production_method = {
					building_type = building_livestock_ranch
					production_method = pm_intensive_grazing_ranch
				}
				}
                activate_production_method = {
				building_type = building_livestock_ranch
				production_method = pm_intensive_grazing_ranch
		        }				
			}
		}
		
		if = {
			limit = {
			    c:PRU ?= {
					is_ai = yes
				}
			}
			c:PRU = {
				s:STATE_POMERANIA = {
					random_scope_state = {
						start_building_construction = building_port
					}
				}
				s:STATE_WESTPHALIA = {
					random_scope_state = {
						start_building_construction = building_tooling_workshops
					}
				}
			}
			c:MEC ?= {
		        set_owes_obligation_to = {
			    	country = c:PRU
			    	setting = yes
			    }
		    }
		}
		if = {
			limit = {
			    c:DEN ?= {
					is_ai = yes
				}
			}
			if = {
			limit = {
				any_scope_state = { # danes are obsessed with luxury furniture
						can_activate_production_method = {
							building_type = building_furniture_manufacturies
							production_method = pm_luxury_furniture
					    }	
					}
			}
			activate_production_method = {
				building_type = building_furniture_manufacturies
				production_method = pm_luxury_furniture
		    }
		}
		}
		if = {
			limit = {
			    c:USA ?= {
					is_ai = yes
				}
			}
			c:USA = {
				random_scope_state = {
					limit = {
						state_region = s:STATE_FLORIDA
						is_incorporated = no
					}
					# Civil war will be a mess without florida
					set_state_type = incorporated
				}
			}
		}
	}
}

vfm_ai.4 = {
	type = country_event
	placement = root

	hidden = yes

	duration = 1

	trigger = {
		is_ai = yes
	}

	immediate = {
	    if = {
			limit = {
			    OR = {
			    AND = {
			        exists = c:CHI
					this = c:CHI
				}
				AND = {
			        exists = c:JAP
					this = c:JAP
				}
				}
				gold_reserves < 1000000 # 1 mil
				scaled_debt <= 0.35
			}
			set_government_wage_level = very_low
			set_military_wage_level = low
		}
	    else_if = {
			limit = {
				scaled_debt > 0.35
			}
			set_government_wage_level = very_low
			set_military_wage_level = very_low
		}
		else_if = {
			limit = {
				scaled_debt > 0.05
			}
			set_government_wage_level = low
			set_military_wage_level = low
		}
		else = {
		    set_government_wage_level = medium
			set_military_wage_level = medium
		}
	}
}

vfm_ai.6 = {
    type = country_event
	placement = ROOT
	#hidden = yes
	
	title = vfm_ai.6.t
	desc = vfm_ai.6.d
	flavor = vfm_ai.6.f
	
	event_image = {
		video = "votp_realist_household"
	}
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"
	
	duration = 1
    
    trigger = {
		is_player = yes
        any_scope_state = {
            NOT = {is_homeland_of_country_cultures = ROOT}
			is_incorporated = no
			is_isolated_from_market = no
			has_decree = decree_greener_grass_campaign
        }
    }  
    immediate = {
        random_scope_state = {
            limit = {
				owner = ROOT
				NOT = {is_homeland_of_country_cultures = ROOT}
				is_incorporated = no
				is_isolated_from_market = no
				has_decree = decree_greener_grass_campaign
            }
            save_scope_as = move_to_state
        }
    }
	
	option = {
		custom_tooltip = vfm_ai.6_tt
		custom_tooltip = greener_grass_campaign_events_2_tt
		name = vfm_ai.6.a
		random_scope_state = { #Move some pops from random incorporated state with >200k population to the choosen unincorporated state
            limit = {
                is_incorporated = yes
                owner = ROOT
                state_population > 500000
                any_scope_culture = {
                    is_primary_culture_of = ROOT
                }
            }
            every_scope_culture = {
                limit = {
                    is_primary_culture_of = ROOT
                }
                ordered_scope_pop = {
                    order_by = total_size
                    move_partial_pop = {
                        state = scope:move_to_state
                        population_ratio = {
                            value = 0.03
                        }
                    }
                }
            }
        }
	}
}
