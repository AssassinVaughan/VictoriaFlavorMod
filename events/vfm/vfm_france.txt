vfm_france.4 = { #Preparation for 1848, swap ideology from liberal to radical
	type = country_event

	title = vfm_france.4.t
	desc = vfm_france.4.d
	flavor = vfm_france.4.f

	event_image = {
		video = "votp_conspiring"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	duration = 3

	trigger = {
	    has_modifier = fra_secular_legacy
		game_date > 1844.1.1	
	}
	
	immediate = {
		ig:ig_intelligentsia = {
			save_scope_as = intelligentsia_ig
		}
	}

	option = { 
		name = vfm_france.4.a
		default_option = yes
		remove_modifier = fra_secular_legacy
		add_modifier = fra_secular_legacy_2
		if = {
			limit = {
			    NOT = { has_global_variable = delescluze_spawn }
				empty_agitator_slots >= 1
			}
			trigger_event = { id = historical_agitators.2 popup = yes }
		}
		if = {
			limit = {
				ig:ig_intelligentsia = {
					leader = {
						NOT = {
							has_ideology = ideology:ideology_radical
						}
					}
				}
	        }
			ig:ig_intelligentsia = {
				leader = {
					set_ideology = ideology:ideology_radical
				}
			}
		}
	}
}
