    selection_behavior = weighted_random

    south_american_caudillo_ruler_vanilla_remover = {
        dna_modifiers = {
            accessory = {
                mode = add
                gene = civilian_coats
                template = no_civilian_coat
                value = 0
            }
        }
        dna_modifiers = {
            accessory = {
                mode = add
                gene = waistcoats
                template = no_waistcoat
                value = 0
            }
        }
        weight = {
            base = 0
            modifier = { #Is a South American
                add = 1
                scope:character ?= {
                    agitator_character_trigger = no
                    south_american_clothes_character_trigger = yes
                    OR = {
                        AND = { #in a Republic
                            republic_clothes_trigger = yes
                            OR = { # Ruler during the Age of Caudillos
                                age_of_caudillos_trigger = yes
                                is_ruler = yes
                            }
                        }
                        AND = { #General during the Age of Caudillos
                            age_of_caudillos_trigger = yes
                            military_clothes_trigger = yes
                        }
                        AND = { #Monarch and not a child
                            age > 12
                            is_ruler = yes
                            monarchy_clothes_trigger = yes
                        }
                    }
                }
            }
        }
    }
}
