potato_blight = {
	trigger = {
		OR = { in_potato_blight_zone = yes in_potato_blight_mild_zone = yes }
	}
	
	incompatible_with = disease_outbreak
	incompatible_with = potato_blight_strong

	range = {
		integer_range = {
			min = 2
			max = 3
		}
	}

	duration = {
		fixed_range = {
			min = 96
			max = 192
		}
	}

	modifier = {
		building_rye_farm_throughput_add = -0.1
	    building_subsistence_farm_throughput_add = -0.1
		state_food_security_add = -0.02
	}

	intensity = {
		fixed_range = {
			min = 3
			max = 5
		}
	}

	chance = {
		value = 0.001
		if = {
			limit = {
				in_potato_blight_zone = yes
			}
			multiply = 4
		}
		if = {
			limit = {
				has_global_variable = hungry_forties_var
			}
			multiply = 25
		}
	}

	color = { 78 70 57 }
	icon = "gfx/interface/icons/harvest_condition_icons/disease_outbreak.dds"
	map_texture = "gfx/map/map_painting/harvest_conditions_map_patterns/harvest_conditions_bad.dds"
	graphics = disease_outbreak
}
