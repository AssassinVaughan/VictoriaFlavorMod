potato_blight = {
	trigger = {
		OR = { in_potato_blight_zone = yes in_potato_blight_mild_zone = yes }
	}
	
	incompatible_with = disease_outbreak
	incompatible_with = potato_blight_strong

	range = {
		integer_range = {
			min = 2
			max = 3
		}
	}

	duration = {
		fixed_range = {
			min = 96
			max = 192
		}
	}

	modifier = {
		building_rye_farm_throughput_add = -0.05
	    building_subsistence_farms_throughput_add = -0.05
		state_peasants_consumption_multiplier_add= 0.03
		state_food_security_add = -0.03
	}

	intensity = {
		fixed_range = {
			min = 7
			max = 12
		}
	}

	chance = {
		value = 0.0005
		if = {
			limit = {
				in_potato_blight_zone = yes
			}
			multiply = 5
		}
		if = {
			limit = {
				has_global_variable = potato_blight_var
			}
			multiply = 40
		}
	}

	color = { 78 70 57 }
	icon = "gfx/interface/icons/harvest_condition_icons/disease_outbreak.dds"
	map_texture = "gfx/map/map_painting/harvest_conditions_map_patterns/harvest_conditions_bad.dds"
	graphics = disease_outbreak
}
