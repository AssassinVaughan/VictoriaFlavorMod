russian_aid_income = {
	if = {
		limit = {
			c:RUS ?= {
				is_country_alive = yes
			}
		}
		if = {
			limit = { c:RUS.relations:ROOT >= relations_threshold:friendly }
			value = 500
		}
		else_if = {
			limit = { c:RUS.relations:ROOT >= relations_threshold:amicable }
			value = 350
		}
		else_if = {
			limit = { c:RUS.relations:ROOT >= relations_threshold:cordial }
			value = 200
		}
		else = {
			value = 100
		}
	}
}

metternich_agenda_value = {
	value = 0
	if = {
		limit = {
			leading_producer_of = g:fine_art
		}
		add = 1
	}
	if = {
		limit = {
			owns_entire_state_region = STATE_WEST_GALICIA
		}
		add = 1
	}
	if = {
		limit = {
			techs_researched >= 52
		}
		add = 1
	}
	if = {
		limit = {
			NOT = {
				any_political_movement = {
					OR = {
						is_political_movement_type = movement_socialist
						is_political_movement_type = movement_anarchist
						is_political_movement_type = movement_communist
						is_political_movement_type = movement_nihilist
						is_political_movement_type = movement_liberal
						is_political_movement_type = movement_india_pan_national
						is_political_movement_type = movement_radical
						is_political_movement_type = movement_labor
						is_political_movement_type = movement_feminist
						is_political_movement_type = movement_land_reform
						is_political_movement_type = movement_anti_slavery
						is_political_movement_type = movement_german_national
						is_political_movement_type = movement_italian_national
					}
					political_movement_support >= 0.05
				}
			}
		}
		add = 1
	}
	if = {
		limit = {
			has_law = law_type:law_autocracy
			has_law = law_type:law_secret_police
			has_law = law_type:law_censorship
		}
		add = 1
	}
	if = {
		limit = {
			institution:institution_police.investment >= 3
		}
		add = 1
	}
	if = {
		limit = {
			institution:institution_home_affairs.investment >= 3
		}
		add = 1
	}
	if = {
		limit = {
			power_bloc ?= {
				num_power_bloc_members >= metternich_bloc_target
			}
		}
		add = 1
	}
	if = {
		limit = {
			power_bloc ?= {
				num_power_bloc_members >= metternich_bloc_target_2
			}
		}
		add = 1
	}
	if = {
		limit = {
			power_bloc ?= {
				current_cohesion_number >= metternich_bloc_cohesion_target
			}
		}
		add = 1
	}
}
