secession_escalation_on_action = {
	effect = {
	    if = {
			limit = {
				ROOT = {
					any_scope_state = {
						is_homeland_of_country_cultures = scope:target
						
						any_scope_culture = {
							is_primary_culture_of = scope:target
							save_temporary_scope_as = discrim_culture
			            }
			            culture_percent_state = {
			                target = scope:discrim_culture
			                value >= 0.4
			            }
						turmoil >= 0.1
					}
				}
			}
			ROOT = {
				every_scope_state = {
					limit = {
						is_homeland_of_country_cultures = scope:target
						
						any_scope_culture = {
							is_primary_culture_of = scope:target
							save_temporary_scope_as = discrim_culture
			            }
			            culture_percent_state = {
			                target = scope:discrim_culture
			                value >= 0.5
			            }
						turmoil >= 0.15
					}
					random_list = {
				    80 = {
					set_state_owner = scope:target
					}
					20 = {
					}
					}
				}
				every_scope_state = {
					limit = {
						is_homeland_of_country_cultures = scope:target
						
						any_scope_culture = {
							is_primary_culture_of = scope:target
							save_temporary_scope_as = discrim_culture
			            }
			            culture_percent_state = {
			                target = scope:discrim_culture
			                value >= 0.3
			            }
						turmoil >= 0.15
					}
					random_list = {
				    50 = {
					set_state_owner = scope:target
					}
					50 = {
					}
					}
				}
			}
		}
		if = {
			limit = {
				ROOT = {
					any_scope_state = {
						is_homeland_of_country_cultures = scope:target
						NOT = { has_claim_by = scope:target }
					}
				}
			}
			ROOT = {
				every_scope_state = {
					limit = {
						is_homeland_of_country_cultures = scope:target
						NOT = { has_claim_by = scope:target }
					}
					state_region = {
						add_claim = scope:target
					}
				}
			}
			#random_diplomatic_play = { # a bit much maybe?
			#	limit = {
			#		any_scope_play_involved = { this = ROOT }
			#		any_scope_play_involved = { this = scope:target }
			#		is_diplomatic_play_type = dp_secession
			#	}
			#	every_state = {
			#		limit = {
			#			owner = ROOT
			#			is_homeland_of_country_cultures = scope:target
			#		}
			#		prev = {
			#			add_war_goal = {
			#				holder = scope:target
			#				type = return_state
			#				target_state = prev
			#			}
			#		}
			#	}
			#}
		}
		
		if = {
			limit = {
			    NOT = {
			        exists = c:CSA
				    scope:target = {
				    	c:CSA ?= this
				    }
				}
			}
		scope:target = {
		    add_modifier = {
			    name = nationalist_fervor
			    months = 25
				is_decaying = yes
		    }
			random_primary_culture = {
				save_temporary_scope_as = discrim_culture
			}
			every_scope_state = {
			every_scope_pop = {
				limit = {
				    NOT = { culture = scope:discrim_culture }
					OR = {
                    	is_pop_type = bureaucrats
                    	is_pop_type = soldiers
                    	is_pop_type = officers
                    	is_pop_type = aristocrats
                   }
				}
				change_pop_culture = {
					target = scope:discrim_culture
					value = 0.5
				}
			}
			}
			add_loyalists = {
			    culture = scope:discrim_culture
		    	pop_type = soldiers
				value = 0.1
		    }
			add_loyalists = {
			    culture = scope:discrim_culture
		    	pop_type = officers
				value = 0.1
		    }
		}
		}
		else_if = {

			    exists = c:USA

grosdeutschland_on_action = {
	effect = {
		if = {
			limit = {
				c:GER ?= ROOT
			}
			trigger_event = { id = german_unification.4 popup = yes }
			trigger_event = { id = grosdeutschland.3 days = 1 }
			if = {
				limit = {
					exists = c:AUS
				}
				c:AUS = {
					trigger_event = { id = grosdeutschland.5 days = 1 }
				}
			}
		}
	}
}

# Fires when a diplomatic play is started
# root = The diplomatic play that has been started
# scope:initiator = Initiator of the play
# scope:target = Target of the play
on_diplomatic_play_started = {
    on_actions = {
		prussian_ai_on_action
		schleswig_holstein_solver_check
	}
}

schleswig_holstein_solver_check = {
	effect = {
		if = {
			limit = { 
			    scope:initiator = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
				}
			    scope:target ?= {
					OR = { c:DEN ?= this c:HOL ?= this c:SCH ?= this }
				}
			}
			scope:initiator = {
				set_variable = {
					name = je_potential_sch_hol_solver 
					years = 4
				}
			}
		}
	}
}

prussian_ai_on_action = {
	effect = {
		if = {
			limit = { 
			    is_diplomatic_play_type = dp_liberate_subject 
				exists = c:MEC
				scope:initiator = {
					OR = { country_has_primary_culture = cu:north_german country_has_primary_culture = cu:south_german }
					is_in_same_power_bloc = c:MEC
					is_ai = yes
				}
				scope:target ?= {
					OR = { c:DEN ?= this c:HOL ?= this c:SCH ?= this }
				}
				c:MEC ?= {
				    NOT = { is_subject_of = scope:initiator }	
				    is_ai = yes
					is_at_war = no
					is_active_in_diplomatic_play = no
		        }
			}
			
			if = { # sanity check, dont drag in UK puppet
				limit = {
			   	    exists = c:HAN
			   	    c:MEC ?= {
						any_scope_treaty = {
							binds = c:HAN
							any_scope_article = {
								has_type = alliance
							}
						}
					}
				}
				every_scope_treaty = {
					limit = {
						binds = c:HAN
						any_scope_article = {
							has_type = alliance
						}
					}
					withdraw = {
						country = root
					}
		   		}
			}
			add_initiator_backers = { c:MEC }
		}
	}
}
