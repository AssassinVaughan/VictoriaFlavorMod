vfm_barracks_colony_created = {
	effect = {
		#temporary solution giving barracks to colonies to prevent being overrun by uprisings instantly
		if = {
			limit = {
				owner = {
					country_is_in_africa = no
					is_ai = yes
				}
				is_under_colonization = yes
				is_incorporated = no
				OR = {
					state_in_africa = yes
				}
				NOT = {
					has_building = building_barracks
				}
			}
			create_building = {
				building = building_barracks
				level = 1
			}
		}
	}
}

revolution_safety_check_on_action = {
	effect = {
		if = {
			limit = {
				total_population <= 100000
				NOT = { has_variable = revolution_safety_check_var }
                is_ai = yes
			}
			set_variable = revolution_safety_check_var
			set_immune_to_revolutions = yes
		}
		else_if = {
			limit = {
				total_population >= 150000
				has_variable = revolution_safety_check_var
			}
			remove_variable = revolution_safety_check_var
			set_immune_to_revolutions = no
		}
	}
}

IQU_joins_secession = {
    effect = {
		if = {
			limit = {
				exists = c:IQU
				scope:target ?= {
					OR = {
			            c:TWT ?= this
			            c:AYM ?= this
			        }
				}
				ROOT = {
					OR = {
			            c:NPU ?= this
			            c:PEU ?= this
			        }
				}
			}
			scope:target ?= {
				set_variable = {
					name = peru_rebel_country_var
					months = short_modifier_time
				}
			}
			c:IQU = {
				trigger_event = { id = peru_bolivia_events.101 days = 5 }
			}
		}
	}	
}
