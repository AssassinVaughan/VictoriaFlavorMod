    group = je_group_internal_affairs
	activation_description = {
		always = yes # TODO PRCAL-45586
	}
	is_shown_when_inactive = {
		country_has_primary_culture = cu:greek
	}

		custom_tooltip = {
			text = greek_enlightenment_state_possible_tt

				state_region = {

				count >= 4
			}
		}
		ig:ig_intelligentsia = {
			is_marginal = no

			is_building_type = building_university
		}
	}

	immediate = {
		trigger_event = { id = vfm_greece.1 }
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header

		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			ig:ig_intelligentsia = {
			is_powerful = yes
		}
		any_scope_building = {
			is_building_type = building_university
			occupancy >= 0.9
			level >= 2
		}
		}
		effect = {
			trigger_event = { id = vfm_greece.2 }
	    }
	}

	weight = 10
	
	should_be_pinned_by_default_uninvolved_or_context = yes
}
