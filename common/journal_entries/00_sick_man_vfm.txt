je_sick_man_arabia = {
	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	is_shown_when_inactive = {
		country_has_primary_culture = cu:turkish

			has_variable = sick_man_complete
		}
	}
	immediate = {
		set_variable = {
			name = sick_man_arabia_var
			value = 5
		}
    }
		
	possible = {
	}
	
	activation_description = {
		always = yes # TODO PRCAL-45586
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
        scope:journal_entry = {
            is_goal_complete = yes
        }
        owns_entire_state_region = STATE_BASRA
        owns_entire_state_region = STATE_BAGHDAD
        
        any_scope_state = {
            state_region = s:STATE_BAGHDAD
            is_incorporated = yes
            turmoil < 0.15
            has_famine = no
            s:STATE_BAGHDAD ?= {
                any_scope_state = { average_sol >= 8 }
            } 
        } 
        any_scope_state = {
            state_region = s:STATE_BASRA
            is_incorporated = yes
            turmoil < 0.15
            has_famine = no
            s:STATE_BASRA ?= {
                any_scope_state = { average_sol >= 8 }
            }
        }
    }

	effect = {
	    trigger_event = { id = ottoman_arabia_events.2 }
		custom_tooltip = je_sick_man_advance
		change_variable = {
			name = sick_man_var
			add = 1
		}
	}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_BAGHDAD
          			    has_famine = no
					    average_sol > 8
					} 
					any_scope_state = {
						state_region = s:STATE_BASRA
        			    has_famine = no
					    average_sol > 8
					}
				}
				change_variable = {
					name = sick_man_arabia_var
					add = 1
				}
			}
			else_if = {
				limit = {
	   				var:sick_man_arabia_var >= 0
				}
			    change_variable = {
					name = sick_man_arabia_var
					subtract = 1
				}
			}
		}
	}
	
    outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = je_standard_failure_conclusion_desc
		effect = {
	    add_fervor_target_modifier = {
			culture = cu:mashriqi
			days = long_modifier_time
			multiplier = 5
		}
	    trigger_event = { id = ottoman_arabia_events.1 }
		}
    }
	
	timeout = 9125

	current_value = {
		value = var:sick_man_arabia_var
	}

	goal_add_value = {
		add = 180
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = finished_sick_man_tt
				has_variable = sick_man_complete
			}
		}
	}

	transferable = no

	weight = 5000
    should_be_pinned_by_default_uninvolved_or_context = yes
	progressbar = yes
}
