ai_strategy_conservative_agenda = {
	icon = "gfx/interface/icons/ai_strategy_icons/conservative_agenda.dds"

	type = political

	revolution_aversion = {
		value = 75
		if = {
			limit = { has_journal_entry = je_tp_pool_suppress_revolts }
			add = 50
		}
	}

	min_law_chance_to_pass = {
		value = 20
		if = {
			limit = { ai_has_reasons_to_not_oppose_law = yes }
			multiply = 0.5
		}
	}

	max_progressiveness = {
		value = 50
		if = {
			limit = { has_enactment_je_or_law_commitment = yes }
			multiply = 2
		}
		if = {
			limit = { any_interest_group = { is_insurrectionary = yes is_powerful = yes } }
			multiply = 1.5
		}
	}
	
	max_regressiveness = {
		value = 25
		if = {
			limit = { any_interest_group = { is_insurrectionary = yes is_powerful = yes } }
			multiply = 1.5
		}
		if = {
			limit = { has_revolutionary_socialist_laws = yes }
			multiply = 2
		}
		if = {
			limit = { has_enactment_je_or_law_commitment = yes }
			multiply = 2
		}
	}

	change_law_chance = {
		value = 2.5

		if = {
			limit = { any_interest_group = { is_insurrectionary = yes } }
			add = 7.5
		}
		
		if = {
			limit = { any_interest_group = { is_insurrectionary = yes is_powerful = yes } }
			add = 7.5
		}
		
		if = {
			limit = { 
			    any_political_movement = {
					political_movement_radicalism >= 70
				}
			}
			add = 7.5
		}

		if = {
			limit = {
				has_revolutionary_socialist_laws = yes
			}
			add = 25
		}

		if = {
			limit = {
				has_enactment_je_or_law_commitment = yes
			}
			add = 25
		}
	}

	pro_interest_groups = {
		ig_rural_folk
		ig_devout
		ig_landowners
		ig_armed_forces
		ig_petty_bourgeoisie
	}

	anti_interest_groups = {
		ig_trade_unions
	}

	pro_movements = {
		movement_royalist_constitutional
		movement_royalist_absolutist
		movement_religious_majority
		movement_cultural_majority
		movement_utilitarian
		movement_corporatist
		movement_modernizer
		movement_orleanist
		movement_bonapartist
	}

	anti_movements = {
		movement_socialist
		movement_labor
		movement_communist
		movement_anarchist
		movement_nihilist
		movement_feminist
		movement_radical
		movement_german_national
		movement_italian_national
		movement_yugoslav_national
	}

	institution_scores = {
		institution_colonial_affairs = {	
			value = 0			
			if = {
				limit = {
					country_rank = rank_value:great_power
                    any_neighbouring_state = {	
					    owner = { is_country_type = decentralized }
                    }					
				}
				add = 110
			}
			else_if = {
				limit = {
				    country_rank >= rank_value:minor_power
					capital = {
						OR = {
							state_is_in_africa = yes
							state_is_in_north_america = yes
							region = sr:region_brazil	
							region = sr:region_la_plata
							region = sr:region_andes	
							region = sr:region_gran_colombia
							region = sr:region_oceania
							region = sr:region_indonesia
							state_region = s:STATE_HOLLAND
							state_region = s:STATE_ESTREMADURA
							state_region = s:STATE_FLANDERS
							state_region = s:STATE_NEW_CASTILE
						}
					}
                    any_neighbouring_state = {	
					    owner = { is_country_type = decentralized }
                    }					
				}
				add = 80
			}
			if = {
				limit = {
					has_journal_entry = je_colonize_korea
				}
				add = 60
			}
            if = {
				limit = {
					has_journal_entry = je_tp_pool_develop_frontier
				}
				add = 60
			}
		}
		institution_social_security = {
			value = 50
		}
		institution_workplace_safety = {
			value = 50
		}
		institution_schools = {
			value = 100
			if = {
				limit = {
					has_journal_entry = je_tp_pool_educational_reform
				}
				multiply = 2
			}
		}
		institution_police = {
			value = 150
		}
		institution_health_system = {
			value = 100
		}
		institution_home_affairs = {
			value = 150
			if = {
				limit = {
					has_journal_entry = je_tp_pool_secret_police
				}
				multiply = 1.5
			}
		}
	}

	possible = {
		always = yes
	}

	weight = {
		value = 20

		if = {
			limit = {
				exists = ruler
				ruler = {
					OR = {
						has_ideology = ideology:ideology_traditionalist
						has_monarchist_ideology = yes
					}
				}
			}

			add = 10
		}

		if = { # Metternich System
			limit = {
				c:AUS ?= this
				has_journal_entry = je_metternich
			}
			add = 20
		}

		if = {
			limit = {
				exists = c:FRA
				c:FRA = this
				ruler = {
					var:dynasty_var ?= flag:house_orleans
				}
			}
			add = 10
		}

		if = {
			limit = {
				is_princely_state = yes
			}
			add = 10
		}

		if = {
			limit = {
				has_radical_leftist_government = yes
			}
			multiply = {
				value = 1
				subtract = {
					value = 1
					subtract = 0.25
					multiply = {
						value = 1
						add = modifier:country_other_country_socialist_government_hostility_impact_mult
