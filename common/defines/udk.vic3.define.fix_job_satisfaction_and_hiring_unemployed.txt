NEconomy = {
	############ Building Hiring Changes ############
	#	These changes exist to fix issue of unable to hire Unemployed Pops when you have too many Building Levels of the Building in same State (Also can happen if you expand a Building too fast)
	#	The "MIN_POPS_PER_PROFESSION" define change is taken from a Paradox Forum Bug Report, where some users noted that increasing this Define's value helped.
	SUBSIDIES_MIN_GAIN = 100							#0		# If a building is subsidized, it will always collect enough subsidies to ensure at least this weekly cash reserve growth
	DEFAULT_MIN_HIRING_RATE = 0.001 					#0.01	# Default weekly min hiring limit applied in a building per profession as a ratio of full employment
	DEFAULT_MAX_HIRING_RATE = 0.1 						#0.1	# Default weekly max hiring limit applied in a building per profession as a ratio of full employment
	HIRING_RANGE_WEEKLY_CHANGE = 0.001					#0.01 	# Weekly hiring range of a building is decreased or increased by this based on whether it is failing to hire at its current rate or not
	MIN_POPS_PER_PROFESSION = 20 						#3		# A building can hire at least this many pops in a profession, even if it has fewer than this number of pops available
}
