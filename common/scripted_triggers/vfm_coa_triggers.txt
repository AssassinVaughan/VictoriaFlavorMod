coa_def_recognized_trigger = {
	scope:actor = {  is_country_type = recognized }
}

REPLACE:coa_def_lesser_subject_trigger = {
	OR = {
		NOT = { exists = scope:actor }
		AND = {
			exists = scope:actor
			scope:actor ?= { 
				OR = {
					is_subject_type = subject_type_colony
					is_subject_type = subject_type_puppet
					is_subject_type = subject_type_vassal
					is_subject_type = subject_type_dependency
				}
			}
		}
	}
}

coa_def_protectorate_trigger = {
	OR = {
		NOT = { exists = scope:target }
		AND = {
			exists = scope:target
			scope:target = { 
				OR = {
					is_subject_type = subject_type_protectorate
				}
			}
		}
	}
}

REPLACE:coa_def_netherlands_ensign_trigger = {
	OR = {
		AND = {
			c:NET ?= { save_temporary_scope_as = custom_overlord }
			coa_def_ensign_trigger = yes
		}
		AND = {
			c:UNL ?= { save_temporary_scope_as = custom_overlord }
			coa_def_ensign_trigger = yes
		}
	}
}

coa_def_hispanic_trigger = {
    any_primary_culture = {
    	has_discrimination_trait = hispanophone
    	has_discrimination_trait = european_heritage
    }
}

REPLACE:coa_SWE_karl_johan_is_king_trigger = {
	exists = c:SWE
	c:SWE ?= {
		NOT = {
			ruler = {
				has_variable = is_johan
			}
		}
	}
}

coa_time_bol_trigger = {
	game_date > 1888.1.1
}

coa_def_time_bol_flag_trigger = {
	scope:actor ?= { coa_time_bol_trigger  = yes }
}

coa_def_argentina_ensign_trigger = {
	exists = scope:actor
	scope:actor = { 
		country_has_primary_culture = cu:platinean
		any_country = {
			has_diplomatic_pact = {
				who = scope:actor
				type = puppet
				is_initiator = no
			}
		}
	 }
}

coa_def_frca_ensign_trigger = {
	exists = c:UCA
	c:UCA = { has_variable = is_frca_var }
	c:UCA ?= { save_temporary_scope_as = custom_overlord }
	coa_def_ensign_trigger = yes
}

coa_def_soconosco_uca_trigger = {
	exists = c:CIP
	c:CIP = { 
		is_subject_of = c:UCA
		NOT = { owns_entire_state_region = STATE_CHIAPAS } 
	}
}

coa_def_soconosco_trigger = {
	exists = c:CIP
	c:CIP = { 
		NOT = { owns_entire_state_region = STATE_CHIAPAS } 
	}
	OR = {
		c:CIP = { is_subject_of = c:ALT }
		c:CIP = { is_subject_of = c:GUA }
	}
}
