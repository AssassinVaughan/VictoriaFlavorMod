expand_madagascar = {
	is_shown = {
		OR = { country_is_in_europe = yes country_rank = rank_value:great_power }
		exists = c:MAD
		c:MAD = {
			is_subject_of = ROOT
		}
	}
	possible = {
		any_state = {
			owner = ROOT
			is_in_geographic_region = geographic_region_zanj_old
			is_under_colonization = no
			state_region = {
				is_homeland = cu:malagasy
			}
		}
	}
	when_taken = {
		custom_tooltip = expand_madagascar_tt
		if = {
			limit = {
				any_state = {
					owner = ROOT
					is_in_geographic_region = geographic_region_zanj_old
					state_region = {
						is_homeland = cu:malagasy
					}
				}
			}
			every_scope_state = {
				limit = {
					owner = ROOT
					is_in_geographic_region = geographic_region_zanj_old
					state_region = {
						is_homeland = cu:malagasy
					}
				}
				set_state_owner = c:MAD
				set_state_type = unincorporated
			}
		}
	}
	
	ai_chance = {
		value = 100
	}
}
