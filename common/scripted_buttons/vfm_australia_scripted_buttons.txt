	name = "je_australia_gbr_revoke_van_diemens_button"
	desc = "je_australia_gbr_revoke_van_diemens_button_desc"
	
	visible = {
		exists = c:NSW
		exists = c:TAS
		c:NSW = {
			is_subject_of = c:GBR
		}
		c:TAS = {
			is_subject_of = c:GBR
		}
	}

	possible = {
	    game_date > 1840.1.1
		c:TAS ?= {
			is_ai = yes
		}
		NOT = { has_variable = recently_revoked_colony }
	}

	effect = {
		c:NSW = {
			annex = c:TAS
		}
		set_variable = {
			name = recently_revoked_colony
			months = 24
		}
	}

	ai_chance = {
		base = 0
		modifier = {
		    trigger = { game_date > 1844.1.1 }
		    add = 1
		}
	}
}
