je_sway_settlers_button = {
	name = {
		triggered_desc = {
			desc = "je_sway_settlers_button_1"
			trigger = {
				NOT = {
					has_variable = swaying_settlers_var
				}
			}
		}
		triggered_desc = {
			desc = "je_sway_settlers_button_2"
			trigger = {
				has_variable = swaying_settlers_var
			}
		}
	}
	desc = {
		triggered_desc = {
			desc = "je_sway_settlers_button_1_desc"
			trigger = {
				NOT = {
					has_variable = swaying_settlers_var
				}
			}
		}
		triggered_desc = {
			desc = "je_sway_settlers_button_2_desc"
			trigger = {
				has_variable = swaying_settlers_var
			}
		}
	}	
	visible = {
		always = yes
	}

	possible = {
	}

	effect = {
		if = {
			limit = {
				has_variable = swaying_settlers_var
			}
			remove_variable = swaying_settlers_var
			remove_modifier = colonial_boom
		}
		else = {
			set_variable = {
				name = swaying_settlers_var
				value = 1
			}
			add_modifier = {
				name = colonial_boom
			}
		}
	}

	ai_chance = {
		value = 0
		#Doesn't have button, is Liberal
		if = {
			limit = {
				AND = {
					has_strategy = ai_strategy_progressive_agenda
					NOT = { has_variable = swaying_settlers_var }
				}
			}
			add = 50
		}
		#Has button, is liberal
		if = {
			limit = {
				AND = {
					has_strategy = ai_strategy_progressive_agenda
					has_variable = swaying_settlers_var
				}
			}
			add = -50
		}
		#Has button, running bureaucracy deficit
		if = {
			limit = {
				AND = {
					bureaucracy < 0
					has_variable = importing_convicts_var
				}
			}
			add = 100
		}
		#Doesn't have button, running bureaucracy deficit
		if = {
			limit = {
				AND = {
					bureaucracy < 0
					NOT = { has_variable = importing_convicts_var }
				}
			}
			add = -100
		}
	}
}

je_australia_gbr_revoke_van_diemens_button = {
	name = "je_australia_gbr_revoke_van_diemens_button"
	desc = "je_australia_gbr_revoke_van_diemens_button_desc"
	
	visible = {
		exists = c:NSW
		exists = c:TAS
		c:NSW = {
			is_subject_of = c:GBR
		}
		c:TAS = {
			is_subject_of = c:GBR
		}
	}

	possible = {
	    game_date > 1840.1.1
		c:TAS ?= {
			is_ai = yes
		}
		NOT = { has_variable = recently_revoked_colony }
	}

	effect = {
		c:NSW = {
			annex = c:TAS
		}
		set_variable = {
			name = recently_revoked_colony
			months = 24
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { game_date > 1844.1.1 }
		    add = 1
		}
	}
}
