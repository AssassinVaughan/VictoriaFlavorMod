investment_charter = {
	type = investment

	icon = "gfx/interface/icons/company_charter_icons/investment_charter.dds"

	cooldown = { years = 5 }

	ai_possible = {
		company_prosperity >= 50
		owner = {
			NOT = {
				has_law_or_variant = law_type:law_isolationism
			}

			OR = {
				any_scope_treaty = {
					any_scope_article = {
						has_type = foreign_investment_rights
						target_country = root.owner
					}
				}
				num_subjects >= 1
			}
		}
	}

	ai_weight = {
        value = 2 #VFM

		if = {
			limit = {
				owner = {
					has_strategy = ai_strategy_economic_imperialism
				}
			}
			multiply = 1.25
		}

		if = {
			limit = {
				owner ?= {
					has_journal_entry = je_zaibatsu
					has_variable = ai_zaibatsu_curtail
				}
				OR = {
					is_company_type = company_type:company_mitsui
					is_company_type = company_type:company_mitsubishi
					is_company_type = company_type:company_sumitomo
					is_company_type = company_type:company_yasuda
				}
			}
			add = -20
		}

		if = {
			limit = {
				owner ?= {
					has_journal_entry = je_zaibatsu
					has_variable = ai_zaibatsu_cement
				}
				OR = {
					is_company_type = company_type:company_mitsui
					is_company_type = company_type:company_mitsubishi
					is_company_type = company_type:company_sumitomo
					is_company_type = company_type:company_yasuda
				}
			}
			add = 10
		}
    }
}
