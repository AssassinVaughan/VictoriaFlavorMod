STATE_OHIO_state_name_assign = {
    if = {
        limit = {
            owner = {
                c:UOM ?= this
            }
        }
        
        set_state_name = STATE_OHIO_michigander
    }
    else = { # switch back to defaults
        reset_state_name = yes
        reset_hub_names = yes
    }
}

STATE_WASHINGTON_state_name_assign = {
	if = {
		limit = {
			owner = {
				c:USA ?= this
			}
		}

        set_state_name = STATE_WASHINGTON_usa

        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_WASHINGTON_city_usa
        }
	}
	# no name reversion effect in an else statement. If the state becomes Washington, it stays Washington.
}

STATE_BRITISH_COLUMBIA_state_name_assign = { # More likely that British Columbia gets its name from conclusion of Oregon JE, but this serves as a one-off fallback
	if = {
		limit = {
			owner = {
				OR = {
					c:GBR ?= this
					is_subject_of = c:GBR
					any_primary_culture = {
						OR = {
							cu:anglo_canadian = this
							cu:franco_canadian = this
						}
					}
				}
			}
			s:STATE_WASHINGTON = {
				any_scope_state = {
					owner = {
						c:USA ?= this
					}
				}
			}
		}

        set_state_name = STATE_BRITISH_COLUMBIA_british
	}
	# no name reversion effect in an else statement. If the state becomes British Columbia, it stays British Columbia.
}

STATE_CALIFORNIA_state_name_assign = { # Take "alta" off if USA owns it, and didn't take baja
    if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = anglophone
                }
                NOT = {
                    any_primary_culture = {
                        has_discrimination_trait = hispanophone
                    }
                }
            }
        }

        set_state_name = STATE_CALIFORNIA_english
        set_hub_names = english
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = hispanophone
                }
                NOT = {
                    any_primary_culture = {
                        has_discrimination_trait = anglophone
                    }
                }
            }
        }

        set_state_name = STATE_CALIFORNIA_spanish
        set_hub_names = spanish
    }
    else = { # switch back to defaults
        reset_state_name = yes
        reset_hub_names = yes
    }
}

STATE_SANTO_DOMINGO_state_name_assign = {
    if = {
        limit = {
            owner = {
                any_primary_culture = {
                    AND = {
                        has_discrimination_trait = francophone
                        has_discrimination_trait = african_heritage
                    }
                }
                NOT = {
                    any_primary_culture = {
                        has_discrimination_trait = hispanophone
                    }
                }
            }
        }
        set_state_name = STATE_SANTO_DOMINGO_haitian
        set_hub_names = haitian
    }
    else_if = { #If not haitian, see if they're francophone
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = francophone
                    NOT = {
                        has_discrimination_trait = african_heritage
                    }
                }
                NOT = {
                    any_primary_culture = {
                        OR = {
                            has_discrimination_trait = hispanophone
                        }
                    }
                }
            }
        }
        
        set_state_name = STATE_SANTO_DOMINGO_french
        set_hub_names = french
    }
    else = { # switch back to defaults
        reset_state_name = yes
        reset_hub_names = yes
    }
}

        if = {
            limit = {

                    OR = {
                        top_overlord ?= {
                            any_primary_culture = {
                                has_discrimination_trait = francophone
                            }
                            any_primary_culture = {
                                NOR = {
                                    culture = cu:afro_antillean
                                    has_discrimination_trait = hispanophone
                                }
                            }

                            is_subject = no

STATE_ONTARIO_state_name_assign = {	# this effect is also covered by completing je_canada_can or je_canada_gbr but I am redundantly covering all bases in a redundant manner
	if = {
		limit = {
			owner = {
				OR = {
					country_has_primary_culture = cu:anglo_canadian
					country_has_primary_culture = cu:franco_canadian
				}
				has_technology_researched = nationalism
			}
		}

        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_ONTARIO_city_ottawa
        }
	}
	# no name reversion effect in an else statement. If the city hub becomes Ottawa, it stays Ottawa
}

STATE_DISTRICT_OF_COLUMBIA_state_name_assign = {
    if = {
        limit = {
            owner = {
                OR = {
                    c:GBR ?= this
                    AND = {
                        exists = top_overlord
                        exists = c:GBR
                        NOT = { c:USA ?= this }
                        top_overlord = c:GBR
                    }
                }
            }
        }

        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_DISTRICT_OF_COLUMBIA_city_uk
        }
    }
    else = { # switch back to defaults
        reset_state_name = yes
        reset_hub_names = yes
    }
}

STATE_URUGUAY_state_name_assign = {
    if = {
		limit = {
            owner = {
				any_primary_culture = {
					has_discrimination_trait = lusophone
				}
				NOT = {
					any_primary_culture = {
						has_discrimination_trait = hispanophone
					}
				}
            }
		
        }
        set_state_name = STATE_URUGUAY_portuguese
        set_hub_names = portuguese
    }
    else_if = {
		limit = {
            owner = {
				any_primary_culture = {
					has_discrimination_trait = hispanophone
				}
				NOR = {
					c:URU ?= this
					was_formed_from = URU
					any_primary_culture = {
						has_discrimination_trait = lusophone
					}
                }
            }
        }
        set_state_name = STATE_URUGUAY_banda_oriental
    }
    else = { # switch back to defaults
        reset_state_name = yes
        reset_hub_names = yes
    }
}
