	city_type = port
	levels_per_mesh = 1

	ai_nationalization_desire = 0.25 # AI is a bit reluctant to privatize government goods

	unlocking_technologies = {
		navigation
	}

	production_method_groups = {
		pmg_base_building_shipyard
	}

	required_construction = construction_cost_high

	potential = {
		is_coastal = yes
	}
	ownership_type = self
	
	ai_value = {
		value = 250

		if = { # Rather import if smaller country
			limit = {
				owner = {
					gdp > 3500000 # 3.5 mil
					NOT = {
						any_scope_building = { 
            				is_building_type = building_shipyard
            				occupancy <= 0.6
       		       		}
					}
				}
			}
			add = 500
		}
		else_if = {
			limit = {
				owner = {
					gdp > 10000000 # 10 mil
				}
			}
			add = 200
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_harbour.dds"
}
