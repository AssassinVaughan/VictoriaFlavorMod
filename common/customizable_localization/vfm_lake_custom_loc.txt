#victoria
vfm_lake_victoria_name = {
	type = country
	text = {
		trigger = { has_global_variable = vfm_lake_victoria_explorer }
		localization_key = VFM_LAKE_VICTORIA_EXPLORER
	}
	text = {
		trigger = {
			has_global_variable = vfm_lake_victoria_ruler
			has_global_variable = vfm_lake_victoria_monarch
		}
		localization_key = VFM_LAKE_VICTORIA_MONARCH
	}
	text = {
		trigger = { has_global_variable = vfm_lake_victoria_ruler }
		localization_key = VFM_LAKE_VICTORIA_RULER
	}
	text = {
		trigger = { has_global_variable = vfm_lake_victoria_nation }
		localization_key = VFM_LAKE_VICTORIA_NATION
	}
	text = {
		localization_key = VFM_LAKE_VICTORIA_NATIVE
	}
}

vfm_lake_victoria_ordinal = {
	type = country
	text = {
		trigger = {
			has_global_variable = vfm_lake_victoria_ordinal
			global_var:vfm_lake_victoria_ordinal >= 4
		}
		localization_key = VFM_LAKE_ORDINAL_4
	}
	text = {
		trigger = {
			has_global_variable = vfm_lake_victoria_ordinal
			global_var:vfm_lake_victoria_ordinal = 3
		}
		localization_key = VFM_LAKE_ORDINAL_3
	}
	text = {
		trigger = {
			has_global_variable = vfm_lake_victoria_ordinal
			global_var:vfm_lake_victoria_ordinal = 2
		}
		localization_key = VFM_LAKE_ORDINAL_2
	}
	text = {
		localization_key = VFM_LAKE_ORDINAL_1
	}
}
